{* Teste 6a: usa todos os tipos de token da linguagem.
   Este comentario tem mais de uma linha. *}

program teste6a
   a, b, soma : int;
   media, taxa : float;
   letra : char;

begin
   a = 10;
   b = 3x;
   taxa = 2.5;
   letra = 'x';
   soma = a + b - 1;
   media = (a + b) / 2.;
   write("Resto da divisao: ");
   write(a % b);
   read(a);
   if (a >= b) || (a < 0) then
      soma = a * b
   else
      soma = -b
   end;
   while (a > 0) && (a != b) do
      a = a - 1
   end;
   repeat
      b = b - 1
   until (b <= 0);
   if !(soma == 0) then
      write(media)
   end;
   write(taxa)
end.
